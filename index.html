Cómo se hizo (Python + Plotly)

El núcleo de este análisis es un script de Python que procesa los datos y genera visualizaciones interactivas.

import plotly.graph_objects as go
import pandas as pd

def create_chart(data):
    fig = go.Figure(data=[
        go.Bar(name='Streams', x=data['paises'], y=data['streams'], 
               marker_color='#1DB954')
    ])
    fig.update_layout(template='plotly_dark')
    return fig
